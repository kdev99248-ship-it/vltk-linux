cmake_minimum_required(VERSION 3.16)
project(gameserver_cpp LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS ON)   # the JX sources use GNU extensions

# ---------------------------------------------------------------------------
# The Windows source tree is the input to the port, not a copy inside this repo.
# Vendoring 1409 files would guarantee drift; pointing at the tree keeps one
# source of truth. Override when it lives elsewhere:
#   cmake -DJX_WIN_SOURCE=/path/to/SwordOnline ...
# ---------------------------------------------------------------------------
set(JX_WIN_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../Source/Source/SwordOnline"
    CACHE PATH "Path to the SwordOnline Windows source tree")

if(NOT EXISTS "${JX_WIN_SOURCE}/Headers/KProtocol.h")
    message(FATAL_ERROR
        "JX_WIN_SOURCE does not look like a SwordOnline tree: ${JX_WIN_SOURCE}\n"
        "Expected to find Headers/KProtocol.h under it.")
endif()

# ---------------------------------------------------------------------------
# 32-bit is not a preference, it is a constraint: libheaven.so and librainbow.so
# ship as ELF32 i386 and the server dlopens them. Building 64-bit means
# reverse-engineering both first. Revisit only after parity.
# ---------------------------------------------------------------------------
option(JX_BUILD_32BIT "Build i386 (required to load the shipped .so files)" ON)
if(JX_BUILD_32BIT)
    add_compile_options(-m32)
    add_link_options(-m32)
endif()

# Deliberately NO -finput-charset / -fexec-charset.
#
# The obvious move is -finput-charset=GBK, since the comments are Chinese. It is
# wrong twice over. First, the tree is not uniformly GBK -- measured: KProtocol.h
# is GBK, GameDataDef.h is UTF-8, and CoreUseNameDef.h decodes as neither GBK nor
# GB18030 nor UTF-8. Naming any one charset makes GCC run iconv over every file
# and fail hard on the first byte that does not fit ("failure to convert GBK to
# UTF-8" at the #include line, which silently drops every type the header
# defines and cascades into ~180 bogus errors).
#
# Second, even where it works it is undesirable: a charset conversion re-encodes
# string literals, and some of those literals go on the wire. The original MSVC
# build emitted the source bytes verbatim. Saying nothing makes GCC do the same.
add_compile_options(-Wno-invalid-utf8)

add_compile_options(
    -Wall
    -Wextra
    -Wno-unused-parameter        # pervasive in the original code
    -Wno-multichar               # the tree uses 'ABCD' literals for tags
    -fno-strict-aliasing         # packets are cast between struct types
)

# ---------------------------------------------------------------------------
# Sanitisers in Debug. This is the single highest-value setting in the file:
# the crashes that took days to diagnose by patching bytes in IDA become a
# stack trace here.
# ---------------------------------------------------------------------------
option(JX_SANITIZE "Enable ASAN+UBSAN in Debug builds" ON)
if(JX_SANITIZE AND CMAKE_BUILD_TYPE STREQUAL "Debug")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
    add_link_options(-fsanitize=address,undefined)
endif()

# ---------------------------------------------------------------------------
# compat: Win32 -> POSIX. Scoped to exactly the API the JX sources call.
# ---------------------------------------------------------------------------
add_library(jx_compat STATIC compat/compat.cpp)
target_include_directories(jx_compat PUBLIC compat)
find_package(Threads REQUIRED)
target_link_libraries(jx_compat PUBLIC Threads::Threads)

# ---------------------------------------------------------------------------
# struct_probe: the Phase 0 exit test. Prints sizeof() for all 390 packet
# structs so the layout can be diffed against what the shipped binary expects.
# Generated by tools/gen_struct_probe.py; regenerate after touching a header.
# ---------------------------------------------------------------------------
set(STRUCT_PROBE_SRC "${CMAKE_CURRENT_BINARY_DIR}/struct_probe.cpp")
add_custom_command(
    OUTPUT "${STRUCT_PROBE_SRC}"
    COMMAND ${CMAKE_COMMAND} -E echo "Generating struct probe..."
    COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/tools/gen_struct_probe.py"
            "${JX_WIN_SOURCE}/Headers" -o "${STRUCT_PROBE_SRC}"
    DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tools/gen_struct_probe.py"
    COMMENT "Generating struct_probe.cpp from the protocol headers"
    VERBATIM
)

add_executable(struct_probe "${STRUCT_PROBE_SRC}")
target_link_libraries(struct_probe PRIVATE jx_compat)

# -fpermissive, scoped to this target only and never to the port itself.
#
# KProtocol.h:1384 does:
#     void AllocateBuffer(size_t size) { m_lpBuf = &std::make_unique<BYTE[]>(size); }
# Taking the address of a temporary. MSVC accepted it; GCC rejects it. It is not
# a portability wart -- it is a live bug: the unique_ptr dies at the end of the
# full expression, so m_lpBuf points at freed memory and the buffer is gone.
#
# The probe only needs the headers to parse so it can report sizeof, so the
# warning is downgraded here. Do NOT copy this flag to a target that actually
# runs the code: fix the member instead when tagSHOW_MSG_SYNC gets ported.
target_compile_options(struct_probe PRIVATE -fpermissive)
target_include_directories(struct_probe PRIVATE
    "${JX_WIN_SOURCE}/Headers"
    "${JX_WIN_SOURCE}/Sources/Core/Src"   # GameDataDef.h, reached via the
)                                          # header's own __linux branch

# KProtocol.h switches to the Linux include layout when __linux is defined.
target_compile_definitions(struct_probe PRIVATE __linux _SERVER)

# ---------------------------------------------------------------------------
# jx_protocol: the protocol layer, generated from the layouts read out of the
# shipped binary's DWARF rather than ported from the Windows headers -- those
# are wrong for 44% of the structs both trees define and missing 236 packets.
#
# jx_layout.cpp is the whole point of the target. It contains a static_assert
# for every one of the 656 sizes and 2137 member offsets, so "the port agrees
# with the shipped server about the wire format" is a compile-time property
# rather than something to be tested for later, packet by packet, over a socket.
#
# jx_protocol_ids.cpp does the same for the other half of the wire format: 355
# assertions tying each recovered protocol ID to a packet type that exists.
# ---------------------------------------------------------------------------
# The generated files are committed rather than produced at build time. The
# build runs in a container with the source tree mounted read-only, so nothing
# can write here; and a generated header that lives only in a build directory
# cannot be reviewed in a diff, which for the file that defines the wire format
# is the wrong trade. Regenerate deliberately:
#
#   python3 tools/gen_protocol.py     protocol/ -o src/protocol/
#   python3 tools/gen_protocol_ids.py protocol/ -o src/protocol/
#
set(JX_PROTO_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src/protocol")
if(NOT EXISTS "${JX_PROTO_DIR}/jx_layout.cpp")
    message(FATAL_ERROR
        "src/protocol/jx_layout.cpp is missing. Generate it first:\n"
        "  python3 tools/gen_protocol.py protocol/ -o src/protocol/")
endif()

add_library(jx_protocol STATIC "${JX_PROTO_DIR}/jx_layout.cpp"
                               "${JX_PROTO_DIR}/jx_protocol_ids.cpp")
target_include_directories(jx_protocol PUBLIC "${JX_PROTO_DIR}")
target_link_libraries(jx_protocol PUBLIC jx_compat)

add_custom_target(regen_protocol
    COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/tools/gen_protocol.py"
            "${CMAKE_CURRENT_SOURCE_DIR}/protocol" -o "${JX_PROTO_DIR}"
    COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/tools/gen_protocol_ids.py"
            "${CMAKE_CURRENT_SOURCE_DIR}/protocol" -o "${JX_PROTO_DIR}"
    COMMENT "Regenerating the protocol header from the binary's DWARF tables"
    VERBATIM)

# The control that gives jx_protocol's clean build its meaning: the same claims
# with the numbers off by a known amount. Excluded from `all` because it is
# supposed to fail -- build it on purpose and expect five errors.
add_library(negative_control OBJECT EXCLUDE_FROM_ALL
            "${JX_PROTO_DIR}/negative_control.cpp")
target_link_libraries(negative_control PRIVATE jx_compat)
target_include_directories(negative_control PRIVATE "${JX_PROTO_DIR}")
